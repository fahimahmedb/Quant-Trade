name: restaurant-stock tests

on:
  push:
    branches: ["**"]
    paths:
      - "restaurant-stock/**"
      - ".github/workflows/restaurant-stock-tests.yml"
  pull_request:
    paths:
      - "restaurant-stock/**"
      - ".github/workflows/restaurant-stock-tests.yml"

defaults:
  run:
    working-directory: restaurant-stock

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: pip install -r requirements-dev.txt

      - name: Install Chromium (NR-12 mobile overflow, F3 comptage hors-ligne)
        run: playwright install --with-deps chromium

      - name: Run tests (unit + HTTP + NR-01..18 + TC-F1/F2/F3)
        run: pytest -v
