a,b=identity.read_text(),action.read_text();results=[]
mutants=[]
for cls in ['IssuerIdentity','IssuerNameInterval','SecurityIdentity','TickerInterval','ListingInterval']:
 start=a.index('class '+cls+':');end=a.find('\n\n@dataclass',start)
 if end==-1:end=len(a)
 section=a[start:end];old='object.__setattr__(self, "evidence_hashes", _validate_lineage(self.evidence_hashes))';assert section.count(old)==1,cls
 mutants.append((cls,identity,a[:start]+section.replace(old,'_validate_lineage(self.evidence_hashes)')+a[end:]))
for name,old in [('action_lineage','        object.__setattr__(self, "evidence_hashes", tuple(self.evidence_hashes))\n'),('action_legs','        object.__setattr__(self, "legs", tuple(self.legs))\n')]:
 assert b.count(old)==1;mutants.append((name,action,b.replace(old,'')))

 ('action_legs_container',action,b,'if not isinstance(self.legs, (list, tuple)):')]:
